@startuml
start
:进入设置页;
:调整玩家数与卧底数\n(受联动范围约束);
:可选：开启难度筛选并选择难度;
:点击“开始游戏”;
:进入主持页并开始异步加载;

if (词库加载/组装成功?) then (是)
  :随机选择词对;
  :生成玩家列表;
  :随机选择卧底索引并分配词汇;
  :currentIndex = 0, 词汇默认隐藏(???);

  while (当前玩家不是最后一位?) is (是)
    :显示“当前玩家：玩家X”;
    :点击“查看词汇”后显示真实词;
    :点击“确认，下一位”;
    :currentIndex++;
    :词汇重新隐藏(???);
  endwhile (否)

  :显示最后一位;
  :点击“查看词汇”后显示真实词;
  :点击“开始游戏”;

  :进入投票页;
  while (未分出胜负?) is (是)
    :选择1名存活玩家;
    :确认淘汰并更新 isAlive=false;
    :检查胜负条件\n(卧底=0 或 平民<=卧底);
  endwhile (否)

  :弹出结算;
  :返回首页 或 再来一局;
else (否)
  :显示加载失败与“重试”;
  :重试加载;
endif
stop
@enduml
